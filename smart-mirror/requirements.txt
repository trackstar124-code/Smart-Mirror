# Web server (app/main.py)
flask

# HTTP requests (modules/weather.py)
requests

# Environment variable loader — reads your .env file for WEATHER_API_KEY
python-dotenv

# ─── Computer Vision / Gesture Detection (modules/gestures.py) ────────────────
#
# We need a GUI-capable OpenCV so cv2.imshow (the gesture preview window) works.
#
#   macOS (dev): pip installs "opencv-python" — its wheel has GUI support (Cocoa).
#   Pi / Linux : the pip ARM wheel is unreliable for GUI, so we install the apt
#                package "python3-opencv" instead (built with GTK). start.sh
#                does this automatically, and the venv sees it via
#                --system-site-packages. That's why the marker below skips pip
#                OpenCV on Linux — installing both would let the pip copy shadow
#                the GTK one and break imshow again.
#
opencv-python; sys_platform == "darwin"

# Numerical arrays — preprocessing tensors for ONNX inference.
numpy

# ONNX Runtime — runs hand-landmark & palm-detector ONNX models.
# Official ARM64 wheels ship for both Apple Silicon and Raspberry Pi 64-bit,
# so no custom build step is needed on the Pi (unlike the old mediapipe dep).
onnxruntime

# ─── Testing ──────────────────────────────────────────────────────────────────
pytest
pytest-mock
